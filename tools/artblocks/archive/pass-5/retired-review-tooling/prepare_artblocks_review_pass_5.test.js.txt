"use strict";

const test = require("node:test");
const assert = require("node:assert/strict");
const fs = require("node:fs/promises");
const path = require("node:path");
const os = require("node:os");
const { createHash } = require("node:crypto");
const { main } = require("./prepare_artblocks_review_pass");
const { planThirdReviewPass, applyThirdReviewPass } = require("./prepare_artblocks_review_pass_3");
const { planFourthReviewPass, applyFourthReviewPass } = require("./prepare_artblocks_review_pass_4");
const { DECISION_INSTRUCTION, EXPECTED_COUNTS, EXPECTED_APPROVED_COUNTS,
  planFifthReviewPass, checkFifthReviewPass, applyFifthReviewPass } = require("./prepare_artblocks_review_pass_5");

const address = "0x0000000000000000000000000000000000000001";
const key = project => `1:${address}:${project}`;
const json = value => Buffer.from(JSON.stringify(value, null, 2) + "\n");
const digest = value => createHash("sha256").update(value).digest("hex");

async function tree(directory) {
  const files = new Map();
  async function visit(relative = "") {
    for (const entry of await fs.readdir(path.join(directory, relative), { withFileTypes: true })) {
      const file = path.join(relative, entry.name);
      if (entry.isDirectory()) await visit(file);
      else files.set(file, await fs.readFile(path.join(directory, file)));
    }
  }
  await visit();
  return files;
}

async function fixture(t) {
  const directory = await fs.mkdtemp(path.join(os.tmpdir(), "artblocks-fifth-review-"));
  t.after(() => fs.rm(directory, { recursive: true, force: true }));
  const options = { reviews: path.join(directory, "reviews"), preview: path.join(directory, "Development", "Good"),
    curation: path.join(directory, "curation.json"), samplesRoot: path.join(directory, "samples"),
    runtimeOutput: path.join(directory, "Development", "review-pass.json"), rejectedIdentity: key(4), pendingIdentity: key(5),
    expectedCounts: { yes: { collections: 1, tokens: 2 }, no: { collections: 1, tokens: 2 }, "double-check": { collections: 1, tokens: 2 }, notes: 0 } };
  const curation = { version: 1, source: { endpoint: "https://example.test/graphql", discoveredAt: "2026-09-06T00:00:00Z" }, collections: {} };
  const first = { version: 2, collections: {} }, second = { version: 3, passID: "pass-2", collections: {} }, catalog = [];
  await fs.mkdir(options.reviews);
  await fs.mkdir(path.join(options.preview, "Tokens"), { recursive: true });
  await fs.mkdir(path.join(options.preview, "Scripts"));
  for (let project = 1; project <= 7; project++) {
    const name = project === 4 ? "aaa" : project === 5 ? "pool party" : `Collection ${project}`, group = project % 2 ? "good" : "ok";
    const samples = [0, 1].map(invocation => ({ tokenId: String(project * 1000000 + invocation), invocation,
      url: `https://example.test/${project}/${invocation}.${invocation ? "mp4" : "png"}`, extension: invocation ? "mp4" : "png" }));
    const entry = { name, artist: "Artist", group, directory: `${name} 目录 `, samples };
    curation.collections[key(project)] = entry;
    first.collections[key(project)] = { name, group, decision: project <= 6 ? "yes" : "mb static", note: null };
    if (project <= 6) second.collections[key(project)] = { name, group,
      decision: project <= 3 ? ["yes", "no", "mb static"][project - 1] : "double-check", note: project === 1 || project === 6 ? "Historical note" : null };
    catalog.unshift({ name, address, chainId: 1, collectionId: `-dev-good-1-${project}`, reviewGroup: group, reviewIdentity: key(project) });
    await fs.writeFile(path.join(options.preview, "Tokens", `${address}-dev-good-1-${project}.json`), json({ items: samples.map(sample => ({ id: sample.tokenId, url: `https://example.test/${sample.tokenId}.png` })) }));
    await fs.writeFile(path.join(options.preview, "Scripts", `${address}-dev-good-1-${project}.json`), json({ value: `Artist source ${name}` }));
    const folder = path.join(options.samplesRoot, group, entry.directory);
    await fs.mkdir(folder, { recursive: true });
    await fs.writeFile(path.join(folder, "manifest.json"), json({ version: 1, identity: key(project),
      project: { chain_id: 1, contract_address: address, project_id: String(project), name, artist_name: "Artist" },
      tokens: samples.map(sample => ({ token: { token_id: sample.tokenId, invocation: sample.invocation },
        download: { sourceURL: sample.url, extension: sample.extension, file: `${sample.tokenId}.${sample.extension}` } })) }));
    for (const sample of samples) await fs.writeFile(path.join(folder, `${sample.tokenId}.${sample.extension}`), `Saved ${sample.tokenId}`);
  }
  const original = json(curation), firstSource = json(first), secondSource = json(second);
  options.expectedCurationSHA256 = digest(original);
  options.firstPass = { expectedSourceSHA256: digest(firstSource), expectedCounts: {
    yes: { collections: 6, tokens: 12 }, no: { collections: 0, tokens: 0 }, "mb static": { collections: 1, tokens: 2 }, notes: 0 } };
  options.thirdPass = { expectedSourceSHA256: digest(secondSource), expectedCounts: {
    yes: { collections: 1, tokens: 2 }, no: { collections: 1, tokens: 2 }, "mb static": { collections: 1, tokens: 2 }, "double-check": { collections: 3, tokens: 6 }, notes: 2 } };
  await fs.writeFile(options.curation, original);
  await fs.writeFile(path.join(options.reviews, "pass-1.json"), firstSource);
  await fs.writeFile(path.join(options.reviews, "pass-2.json"), secondSource);
  await fs.writeFile(path.join(options.preview, "items.json"), json(catalog));
  await applyThirdReviewPass(await planThirdReviewPass({ ...options, ...options.thirdPass }));
  options.expectedPreviousCurationSHA256 = digest(await fs.readFile(options.curation));
  await applyFourthReviewPass(await planFourthReviewPass(options));
  options.fourthPass = { expectedCounts: options.expectedCounts };
  options.expectedCounts = { yes: { collections: 1, tokens: 2 }, notes: 0 };
  options.expectedApprovedCounts = { collections: 3, tokens: 6 };
  return { directory, options, curation };
}

test("the final review records the literal approval instruction and exact cumulative totals", () => {
  assert.deepEqual(EXPECTED_COUNTS, { yes: { collections: 1, tokens: 20 }, notes: 0 });
  assert.deepEqual(EXPECTED_APPROVED_COUNTS, { collections: 292, tokens: 6524 });
  assert.match(DECISION_INSTRUCTION, /^ok let's mark poll party as yes now\./u);
  assert.match(DECISION_INSTRUCTION, /reset all decisions and notes/u);
});

test("read-only default plans a clean final pass without replacing historical decisions or media paths", async t => {
  const f = await fixture(t), before = await tree(f.directory), logs = [];
  const plan = await main([], { ...f.options, log: value => logs.push(value) });
  assert.deepEqual(await tree(f.directory), before);
  assert.equal(JSON.parse(logs[0]).mode, "dry-run");
  assert.deepEqual(plan.removals, []);
  const source = JSON.parse(plan.files.get(path.join(f.options.reviews, "pass-4-decisions.json")));
  assert.equal(source.passID, "pass-4");
  assert.equal(source.sourceType, "recorded-user-instruction");
  assert.equal(source.deviceExport, false);
  assert.equal(source.instruction, DECISION_INSTRUCTION);
  assert.deepEqual(source.collections, { [key(5)]: { name: "pool party", group: "good", decision: "yes", note: null } });
  assert.deepEqual(source.counts, { yes: { collections: 1, tokens: 2 }, notes: 0 });
  const approved = JSON.parse(plan.files.get(path.join(f.options.reviews, "approved.json")));
  const previous = JSON.parse(before.get(path.join("reviews", "approved.json")));
  assert.deepEqual(approved.collections.map(record => record.identity), [key(6), key(5), key(1)]);
  assert.deepEqual(approved.collections.filter(record => record.identity !== key(5)), previous.collections);
  assert.equal(approved.collections[1].sourcePassID, "pass-4");
  assert.equal(approved.collectionCount, 3);
  assert.equal(approved.tokenCount, 6);
  assert.ok(approved.collections.every(record => record.samples[1].localPath.endsWith(".mp4") && record.samples[1].fallbackImageURL.endsWith(".png")));
  assert.deepEqual(JSON.parse(plan.files.get(f.options.runtimeOutput)), { version: 1, passID: "pass-5",
    sourceReviewSHA256: plan.sourceReviewSHA256, selectedIdentities: [key(6), key(5), key(1)], initialNotesByIdentity: {} });
  assert.equal(plan.files.has(f.options.curation), false);
  assert.equal(plan.files.has(path.join(f.options.reviews, "deferred-static.json")), false);
});

test("publication archives old approvals exactly, preserves curation and samples, and is idempotent", async t => {
  const f = await fixture(t), before = await tree(f.directory), plan = await planFifthReviewPass(f.options);
  assert.equal(await applyFifthReviewPass(plan), 8);
  await checkFifthReviewPass(plan);
  const after = await tree(f.directory);
  const mutable = new Set([path.join("Development", "review-pass.json"), path.join("reviews", "approved.json"), path.join("reviews", "approved.md")]);
  for (const [file, contents] of before) if (!mutable.has(file)) assert.deepEqual(after.get(file), contents, file);
  for (const [archive, original] of [["pass-4-runtime.json", path.join("Development", "review-pass.json")],
    ["pass-4-approved.json", path.join("reviews", "approved.json")], ["pass-4-approved.md", path.join("reviews", "approved.md")],
    ["pass-4-curation.json", "curation.json"]]) {
    assert.deepEqual(after.get(path.join("reviews", archive)), before.get(original));
  }
  assert.equal(await applyFifthReviewPass(await planFifthReviewPass(f.options)), 0);
  assert.deepEqual(await tree(f.directory), after);
  const historyOptions = { ...f.options, ...f.options.fourthPass,
    curation: undefined, runtimeOutput: undefined, log: () => {} };
  await main(["--pass", "pass-4", "--check"], historyOptions);
  await main(["--pass", "pass-4", "--apply"], historyOptions);
  await main(["--pass", "pass-3", "--check"], { ...historyOptions, ...f.options.thirdPass });
  await main(["--pass", "pass-3", "--apply"], { ...historyOptions, ...f.options.thirdPass });
  assert.deepEqual(await tree(f.directory), after);
});

test("unrelated active decisions, changed inputs, and frozen archive edits prevent publication", async t => {
  const f = await fixture(t), active = await fs.readFile(f.options.runtimeOutput);
  await fs.writeFile(f.options.runtimeOutput, json({ ...JSON.parse(active), passID: "unexpected-pass" }));
  await assert.rejects(planFifthReviewPass(f.options), /unrelated changes/u);
  await fs.writeFile(f.options.runtimeOutput, active);
  const plan = await planFifthReviewPass(f.options);
  await fs.appendFile(path.join(f.options.reviews, "deferred-static.md"), "Unrelated edit");
  const before = await tree(f.directory);
  await assert.rejects(applyFifthReviewPass(plan), /changed after validation/u);
  assert.deepEqual(await tree(f.directory), before);
});

test("recorded approval cannot overwrite a different frozen instruction or curation", async t => {
  const f = await fixture(t), source = path.join(f.options.reviews, "pass-4-decisions.json");
  await fs.writeFile(source, "A different instruction");
  await assert.rejects(planFifthReviewPass(f.options), /Refusing to replace frozen review archive/u);
  await fs.rm(source);
  await applyFifthReviewPass(await planFifthReviewPass(f.options));
  const live = JSON.parse(await fs.readFile(f.options.curation));
  live.collections[key(6)].name = "Unrelated curation change";
  await fs.writeFile(f.options.curation, json(live));
  await assert.rejects(planFifthReviewPass(f.options), /Live curation differs/u);
});
