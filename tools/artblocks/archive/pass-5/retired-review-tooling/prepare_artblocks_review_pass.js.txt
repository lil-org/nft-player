#!/usr/bin/env node
"use strict";

const fs = require("node:fs/promises");
const path = require("node:path");
const { createHash, randomUUID } = require("node:crypto");
const { GROUPS, identity, validateCuration } = require("./artblocks/curation");

const ROOT = path.resolve(__dirname, "..");
const DEFAULT_ARCHIVE = path.join(ROOT, "tools", "artblocks", "reviews", "pass-1.json");
const DEFAULT_STATIC_OUTPUT = path.join(ROOT, "tools", "artblocks", "reviews", "pass-1-static.json");
const DEFAULT_PREVIEW = path.join(ROOT, "Suggested Items", "Suggested.bundle", "Development", "Good");
const DEFAULT_RUNTIME_OUTPUT = path.join(ROOT, "tools", "artblocks", "reviews", "pass-2-runtime.json");
const SOURCE_REVIEW_SHA256 = "076a3b03c9a09987e9c1a4e1feca664231219f12794db8c5feaa2c4d188b2689";
const EXPECTED_COUNTS = Object.freeze({
  yes: { collections: 315, tokens: 7028 },
  no: { collections: 104, tokens: 2316 },
  "mb static": { collections: 100, tokens: 2283 },
  notes: 4,
});

const object = value => value !== null && typeof value === "object" && !Array.isArray(value);
const sha256 = value => createHash("sha256").update(value).digest("hex");
const json = value => Buffer.from(JSON.stringify(value, null, 2) + "\n");

function canonicalIdentity(key) {
  const components = key.split(":");
  if (components.length !== 3 || identity(...components) !== key) {
    throw new Error(`Invalid canonical review identity: ${key}`);
  }
}

function pngURL(value, key, tokenId) {
  let url;
  try { url = new URL(value); } catch { throw new Error(`Missing PNG fallback: ${key}/${tokenId}`); }
  if (typeof value !== "string" || value !== value.trim() || url.protocol !== "https:"
    || url.username || url.password || !url.pathname.toLowerCase().endsWith(".png")) {
    throw new Error(`Invalid HTTPS PNG fallback: ${key}/${tokenId}`);
  }
  return value;
}

function createArtifacts(source, curation, catalog, tokensByIdentity, options = {}) {
  const sourceReviewSHA256 = sha256(source);
  const expectedHash = options.expectedSourceSHA256 === undefined ? SOURCE_REVIEW_SHA256 : options.expectedSourceSHA256;
  const review = JSON.parse(source.toString("utf8"));
  if (!object(review) || review.version !== 2 || !object(review.collections)) {
    throw new Error("Expected a version 2 collection review export");
  }
  validateCuration(curation);
  const retained = new Map(Object.entries(curation.collections).filter(([, item]) => GROUPS.includes(item.group)));
  if (!Array.isArray(catalog) || catalog.length !== retained.size) {
    throw new Error("Preview catalog does not match retained curation coverage");
  }
  const catalogByIdentity = new Map();
  for (const item of catalog) {
    if (!object(item) || typeof item.reviewIdentity !== "string") throw new Error("Preview item is missing its review identity");
    const key = item.reviewIdentity;
    canonicalIdentity(key);
    if (!retained.has(key) || catalogByIdentity.has(key)) throw new Error(`Unknown or duplicate preview identity: ${key}`);
    const expected = retained.get(key);
    const [chain, address, project] = key.split(":");
    if (item.address !== address || item.chainId !== Number(chain) || item.collectionId !== `-dev-good-${chain}-${project}`
      || item.name !== expected.name || item.reviewGroup !== expected.group) {
      throw new Error(`Preview metadata differs from curation: ${key}`);
    }
    catalogByIdentity.set(key, item);
  }

  const decisions = ["yes", "no", "mb static"];
  for (const [key, record] of Object.entries(review.collections)) {
    canonicalIdentity(key);
    if (!retained.has(key) || !catalogByIdentity.has(key)) throw new Error(`Unknown review identity: ${key}`);
    const expected = retained.get(key);
    if (!object(record) || record.name !== expected.name || record.group !== expected.group) {
      throw new Error(`Review metadata differs from curation: ${key}`);
    }
    if (!decisions.includes(record.decision)) throw new Error(`Missing or invalid review decision: ${key}`);
    if (!Object.hasOwn(record, "note") || record.note !== null && typeof record.note !== "string") {
      throw new Error(`Invalid review note: ${key}`);
    }
  }
  for (const key of retained.keys()) {
    if (!Object.hasOwn(review.collections, key)) throw new Error(`Missing review identity: ${key}`);
  }

  const selectedIdentities = [];
  const initialNotesByIdentity = {};
  const deferred = [];
  const counts = Object.fromEntries(decisions.map(decision => [decision, { collections: 0, tokens: 0 }]));
  counts.notes = 0;
  for (const item of catalog) {
    const key = item.reviewIdentity;
    const record = review.collections[key];
    const expected = retained.get(key);
    const tokens = tokensByIdentity.get(key)?.items;
    if (!Array.isArray(tokens) || tokens.length !== expected.samples.length
      || tokens.some((token, index) => !object(token) || token.id !== expected.samples[index].tokenId)) {
      throw new Error(`Bundled token selection differs from curation: ${key}`);
    }
    tokens.forEach(token => pngURL(token.url, key, token.id));
    counts[record.decision].collections++;
    counts[record.decision].tokens += tokens.length;
    if (record.note !== null) counts.notes++;
    if (record.decision === "yes") {
      selectedIdentities.push(key);
      if (record.note !== null) initialNotesByIdentity[key] = record.note;
    } else if (record.decision === "mb static") {
      const localCollectionFolder = path.posix.join("samples", expected.group, expected.directory);
      deferred.push({
        identity: key,
        name: record.name,
        group: record.group,
        decision: record.decision,
        note: record.note,
        localCollectionFolder,
        samples: tokens.map((token, index) => ({
          tokenId: token.id,
          localPath: path.posix.join(localCollectionFolder, `${token.id}.${expected.samples[index].extension}`),
          fallbackImageURL: token.url,
        })),
      });
    }
  }
  const expectedCounts = options.expectedCounts ?? EXPECTED_COUNTS;
  for (const decision of decisions) {
    if (counts[decision].collections !== expectedCounts[decision].collections
      || counts[decision].tokens !== expectedCounts[decision].tokens) {
      throw new Error(`Unexpected ${decision} partition: ${JSON.stringify(counts[decision])}`);
    }
  }
  if (counts.notes !== expectedCounts.notes) throw new Error(`Unexpected note count: ${counts.notes}`);
  if (expectedHash !== null && sourceReviewSHA256 !== expectedHash) throw new Error("The first review export SHA-256 does not match the frozen source");

  return {
    archive: Buffer.from(source),
    runtime: {
      version: 1,
      passID: "pass-2",
      sourceReviewSHA256,
      selectedIdentities,
      initialNotesByIdentity,
    },
    deferred: {
      version: 1,
      sourcePassID: "pass-1",
      sourceReviewSHA256,
      collectionCount: counts["mb static"].collections,
      tokenCount: counts["mb static"].tokens,
      collections: deferred,
    },
    counts,
  };
}

async function optionalFile(file) {
  try { return await fs.readFile(file); }
  catch (error) { if (error.code === "ENOENT") return null; throw error; }
}

async function planReviewPass(options = {}) {
  const archive = path.resolve(options.archive ?? DEFAULT_ARCHIVE);
  const preview = path.resolve(options.preview ?? DEFAULT_PREVIEW);
  const [source, curationText, catalogText] = await Promise.all([
    fs.readFile(options.source ?? archive),
    fs.readFile(options.curation ?? path.join(ROOT, "tools", "artblocks", "reviews", "pass-1-curation.json"), "utf8"),
    fs.readFile(path.join(preview, "items.json"), "utf8"),
  ]);
  const catalog = JSON.parse(catalogText);
  if (!Array.isArray(catalog)) throw new Error("Expected a preview catalog array");
  const tokensByIdentity = new Map(await Promise.all(catalog.map(async item => {
    if (!object(item) || typeof item.address !== "string" || !/^0x[\da-f]{40}$/u.test(item.address)
      || typeof item.collectionId !== "string" || !/^-dev-good-\d+-\d+$/u.test(item.collectionId)) {
      throw new Error("Invalid preview token resource identity");
    }
    const file = path.join(preview, "Tokens", `${item.address}${item.collectionId}.json`);
    return [item.reviewIdentity, JSON.parse(await fs.readFile(file, "utf8"))];
  })));
  const artifacts = createArtifacts(source, JSON.parse(curationText), catalog, tokensByIdentity, options);
  const files = new Map([
    [archive, artifacts.archive],
    [path.resolve(options.staticOutput ?? DEFAULT_STATIC_OUTPUT), json(artifacts.deferred)],
    [path.resolve(options.runtimeOutput ?? DEFAULT_RUNTIME_OUTPUT), json(artifacts.runtime)],
  ]);
  if (files.size !== 3) throw new Error("Review output paths must be distinct");
  const previousArchive = await optionalFile(archive);
  if (previousArchive && !previousArchive.equals(artifacts.archive)) throw new Error("Refusing to replace the frozen first review archive");
  return { files, archive, counts: artifacts.counts, sourceReviewSHA256: artifacts.runtime.sourceReviewSHA256 };
}

async function checkReviewPass(plan) {
  const differences = [];
  for (const [file, expected] of plan.files) {
    const actual = await optionalFile(file);
    if (!actual?.equals(expected)) differences.push(file);
  }
  if (differences.length) throw new Error(`Review artifacts are missing or stale: ${differences.join(", ")}`);
}

async function applyReviewPass(plan) {
  const pending = [];
  try {
    for (const [file, contents] of plan.files) {
      const previous = await optionalFile(file);
      if (previous?.equals(contents)) continue;
      if (file === plan.archive && previous) throw new Error("Refusing to replace the frozen first review archive");
      await fs.mkdir(path.dirname(file), { recursive: true });
      const staging = `${file}.${randomUUID()}.tmp`;
      pending.push({ file, staging, contents });
      await fs.writeFile(staging, contents, { flag: "wx" });
    }
    for (const { file, staging, contents } of pending) {
      if (file === plan.archive) {
        try { await fs.link(staging, file); }
        catch (error) {
          if (error.code !== "EEXIST" || !(await fs.readFile(file)).equals(contents)) throw error;
        }
      } else {
        await fs.rename(staging, file);
      }
    }
  } finally {
    await Promise.all(pending.map(({ staging }) => fs.rm(staging, { force: true })));
  }
  return pending.length;
}

async function main(argv = process.argv.slice(2), options = {}) {
  let mode = "dry-run";
  let modeSet = false;
  for (let index = 0; index < argv.length; index++) {
    const argument = argv[index];
    if (["--dry-run", "--apply", "--check"].includes(argument)) {
      if (modeSet) throw new Error("Choose only one review preparation operation");
      mode = argument.slice(2);
      modeSet = true;
    } else if (argument === "--source") {
      const value = argv[++index];
      if (!value || value.startsWith("--")) throw new Error("Missing value for --source");
      options.source = path.resolve(value);
    } else if (argument === "--pass") {
      const value = argv[++index];
      if (!["pass-2", "pass-3", "pass-4", "pass-5"].includes(value)) throw new Error("--pass must be pass-2, pass-3, pass-4, or pass-5");
      options.passID = value;
    } else if (argument === "--help") {
      (options.log ?? console.log)("Usage: node tools/prepare_artblocks_review_pass.js [--dry-run|--apply|--check] [--source FILE] [--pass pass-2|pass-3|pass-4|pass-5]\nPrepares the final 292-collection fifth review pass offline by default. Apply archives pool party's approval and preserves historical decisions without deleting samples. Historical operations use archived outputs without changing the active pass or cumulative approvals.");
      return;
    } else {
      throw new Error(`Unknown option: ${argument}`);
    }
  }
  const passID = options.passID ?? "pass-5";
  let prepare, apply, check;
  if (passID === "pass-2") {
    [prepare, apply, check] = [planReviewPass, applyReviewPass, checkReviewPass];
  } else if (passID === "pass-3") {
    const third = require("./prepare_artblocks_review_pass_3");
    const reviews = path.resolve(options.reviews ?? path.join(ROOT, "tools", "artblocks", "reviews"));
    options = { ...options,
      curation: options.curation ?? path.join(reviews, "pass-3-curation.json"),
      runtimeOutput: options.runtimeOutput ?? path.join(reviews, "pass-3-runtime.json"),
    };
    [prepare, apply, check] = [third.planThirdReviewPass, third.applyThirdReviewPass, third.checkThirdReviewPass];
  } else if (passID === "pass-4") {
    const fourth = require("./prepare_artblocks_review_pass_4");
    const reviews = path.resolve(options.reviews ?? path.join(ROOT, "tools", "artblocks", "reviews"));
    options = { ...options,
      curation: options.curation ?? path.join(reviews, "pass-4-curation.json"),
      runtimeOutput: options.runtimeOutput ?? path.join(reviews, "pass-4-runtime.json"),
      approvedOutput: options.approvedOutput ?? path.join(reviews, "pass-4-approved.json"),
      approvedListOutput: options.approvedListOutput ?? path.join(reviews, "pass-4-approved.md"),
    };
    [prepare, apply, check] = [fourth.planFourthReviewPass, fourth.applyFourthReviewPass, fourth.checkFourthReviewPass];
  } else {
    const fifth = require("./prepare_artblocks_review_pass_5");
    [prepare, apply, check] = [fifth.planFifthReviewPass, fifth.applyFifthReviewPass, fifth.checkFifthReviewPass];
  }
  const plan = await prepare(options);
  if (mode === "apply") await apply(plan);
  if (mode === "check") await check(plan);
  (options.log ?? console.log)(JSON.stringify({ mode, sourceReviewSHA256: plan.sourceReviewSHA256, counts: plan.counts, ...(plan.approvedCounts ? { approvedCounts: plan.approvedCounts } : {}), files: [...plan.files.keys()] }, null, 2));
  return plan;
}

module.exports = { SOURCE_REVIEW_SHA256, EXPECTED_COUNTS, createArtifacts, planReviewPass, checkReviewPass, applyReviewPass, main };
if (require.main === module) main().catch(error => { console.error(error.message); process.exitCode = 1; });
