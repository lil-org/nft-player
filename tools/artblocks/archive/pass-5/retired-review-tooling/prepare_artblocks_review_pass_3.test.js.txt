"use strict";

const test = require("node:test");
const assert = require("node:assert/strict");
const fs = require("node:fs/promises");
const path = require("node:path");
const os = require("node:os");
const { createHash } = require("node:crypto");
const { createArtifacts, main } = require("./prepare_artblocks_review_pass");
const { SOURCE_SHA256, CURATION_SHA256, EXPECTED_COUNTS, planThirdReviewPass, checkThirdReviewPass, applyThirdReviewPass }
  = require("./prepare_artblocks_review_pass_3");

const address = "0x0000000000000000000000000000000000000001";
const key = project => `1:${address}:${project}`;
const json = value => Buffer.from(JSON.stringify(value, null, 2) + "\n");
const digest = value => createHash("sha256").update(value).digest("hex");

async function fixture(t) {
  const directory = await fs.mkdtemp(path.join(os.tmpdir(), "artblocks-third-review-"));
  t.after(() => fs.rm(directory, { recursive: true, force: true }));
  const config = {
    passID: "pass-3", reviews: path.join(directory, "reviews"), preview: path.join(directory, "Development", "Good"),
    curation: path.join(directory, "curation.json"), source: path.join(directory, "export.json"),
    samplesRoot: path.join(directory, "samples"), runtimeOutput: path.join(directory, "Development", "review-pass.json"),
    expectedCounts: { yes: { collections: 1, tokens: 2 }, no: { collections: 1, tokens: 2 },
      "mb static": { collections: 1, tokens: 2 }, "double-check": { collections: 1, tokens: 2 }, notes: 2 },
  };
  const curation = { version: 1, source: { endpoint: "https://example.test/graphql", discoveredAt: "2026-09-06T00:00:00Z" }, collections: {} };
  const first = { version: 2, collections: {} }, second = { version: 3, passID: "pass-2", collections: {} };
  const catalog = [], tokens = new Map();
  for (let project = 1; project <= 6; project++) {
    const group = project % 2 ? "good" : "hmm", name = `Collection ${project}`;
    const samples = [0, 1].map(invocation => ({ tokenId: String(project * 1000000 + invocation), invocation,
      url: `https://example.test/${project}/${invocation}.${invocation ? "mp4" : "png"}`, extension: invocation ? "mp4" : "png" }));
    const entry = { name, artist: "Artist", group, directory: `${name} 目录 `, samples };
    curation.collections[key(project)] = entry;
    first.collections[key(project)] = { name, group, decision: project <= 4 ? "yes" : project === 5 ? "no" : "mb static", note: null };
    if (project <= 4) second.collections[key(project)] = { name, group,
      decision: ["yes", "no", "mb static", "double-check"][project - 1], note: project === 1 || project === 4 ? "Keep historically\n新しいメモ" : null };
    catalog.unshift({ name, address, chainId: 1, collectionId: `-dev-good-1-${project}`, reviewGroup: group, reviewIdentity: key(project) });
    tokens.set(key(project), { items: samples.map(sample => ({ id: sample.tokenId, url: `https://example.test/${sample.tokenId}.png` })) });
    const folder = path.join(config.samplesRoot, group, entry.directory);
    await fs.mkdir(folder, { recursive: true });
    await fs.writeFile(path.join(folder, "manifest.json"), json({ version: 1, identity: key(project),
      project: { chain_id: 1, contract_address: address, project_id: String(project), name, artist_name: "Artist" },
      tokens: samples.map(sample => ({ token: { token_id: sample.tokenId, invocation: sample.invocation },
        download: { sourceURL: sample.url, extension: sample.extension, file: `${sample.tokenId}.${sample.extension}` } })),
    }));
    for (const sample of samples) await fs.writeFile(path.join(folder, `${sample.tokenId}.${sample.extension}`), `Saved ${sample.tokenId}`);
  }
  curation.collections[key(99)] = { name: "Original excluded", artist: "Artist", group: "excluded", directory: "excluded" };
  const originalCuration = json(curation), firstSource = json(first), secondSource = Buffer.from(JSON.stringify(second, null, 4).replaceAll("\n", "\r\n") + "\r\n");
  config.expectedCurationSHA256 = digest(originalCuration);
  config.expectedSourceSHA256 = digest(secondSource);
  config.firstPass = { expectedSourceSHA256: digest(firstSource),
    expectedCounts: { yes: { collections: 4, tokens: 8 }, no: { collections: 1, tokens: 2 }, "mb static": { collections: 1, tokens: 2 }, notes: 0 } };
  const firstArtifacts = createArtifacts(firstSource, curation, catalog, tokens, config.firstPass);
  await fs.mkdir(config.reviews);
  await fs.mkdir(path.join(config.preview, "Tokens"), { recursive: true });
  await fs.mkdir(path.join(config.preview, "Scripts"));
  await fs.writeFile(config.curation, originalCuration);
  await fs.writeFile(config.source, secondSource);
  await fs.writeFile(path.join(config.reviews, "pass-1.json"), firstSource);
  await fs.writeFile(path.join(config.reviews, "pass-1-static.json"), json(firstArtifacts.deferred));
  await fs.writeFile(config.runtimeOutput, json(firstArtifacts.runtime));
  await fs.writeFile(path.join(config.preview, "items.json"), json(catalog));
  for (const item of catalog) {
    await fs.writeFile(path.join(config.preview, "Tokens", `${item.address}${item.collectionId}.json`), json(tokens.get(item.reviewIdentity)));
    await fs.writeFile(path.join(config.preview, "Scripts", `${item.address}${item.collectionId}.json`), json({ value: `Artist source ${item.name}` }));
  }
  return { directory, config, curation, second, originalCuration, firstSource, secondSource };
}

async function tree(directory) {
  const files = new Map();
  async function visit(relative = "") {
    for (const entry of await fs.readdir(path.join(directory, relative), { withFileTypes: true })) {
      const file = path.join(relative, entry.name);
      if (entry.isDirectory()) await visit(file);
      else files.set(file, await fs.readFile(path.join(directory, file)));
    }
  }
  await visit();
  return files;
}

test("the third pass pins the exact export, curation, and 315-collection partition", () => {
  assert.equal(SOURCE_SHA256, "c95cda472c92db8e6bb7afc1d667aade0cbca41c59f0cf3bac1e8898d1c76916");
  assert.equal(CURATION_SHA256, "eff11e853fb991e49ffbb76771eeff5af415a6afa008544c29a0b1d3be4d8cba");
  assert.deepEqual(EXPECTED_COUNTS, { yes: { collections: 214, tokens: 4753 }, no: { collections: 13, tokens: 278 },
    "mb static": { collections: 9, tokens: 203 }, "double-check": { collections: 79, tokens: 1794 }, notes: 13 });
});

test("planning is read-only, preserves historical notes, and starts only double-check collections clean", async t => {
  const f = await fixture(t), before = await tree(f.directory), logs = [];
  const plan = await main([], { ...f.config, log: value => logs.push(value) });
  assert.equal(JSON.parse(logs[0]).mode, "dry-run");
  assert.deepEqual(await tree(f.directory), before);
  assert.deepEqual(plan.files.get(path.join(f.config.reviews, "pass-2.json")), f.secondSource);
  assert.deepEqual(plan.files.get(path.join(f.config.reviews, "pass-1-curation.json")), f.originalCuration);
  const runtime = JSON.parse(plan.files.get(f.config.runtimeOutput));
  assert.deepEqual(runtime, { version: 1, passID: "pass-3", sourceReviewSHA256: digest(f.secondSource), selectedIdentities: [key(4)], initialNotesByIdentity: {} });
  const approved = JSON.parse(plan.files.get(path.join(f.config.reviews, "pass-2-approved.json")));
  assert.equal(approved.collectionCount, 1);
  assert.equal(approved.collections[0].note, f.second.collections[key(1)].note);
  const deferred = JSON.parse(plan.files.get(path.join(f.config.reviews, "deferred-static.json")));
  assert.deepEqual(deferred.collections.map(record => record.identity), [key(6), key(3)]);
  assert.deepEqual(deferred.collections.map(record => record.sourcePassID), ["pass-1", "pass-2"]);
  assert.equal(deferred.tokenCount, 4);
  assert.equal(deferred.collections.flatMap(record => record.samples).filter(sample => sample.localPath.endsWith(".mp4")).length, 2);
  assert.ok(deferred.collections.flatMap(record => record.samples).every(sample => sample.fallbackImageURL.endsWith(".png")));
  await assert.rejects(checkThirdReviewPass(plan), /missing or stale/u);
});

test("apply archives before deleting only newly rejected folders and is idempotent", async t => {
  const f = await fixture(t), before = await tree(f.directory), plan = await planThirdReviewPass(f.config);
  await applyThirdReviewPass(plan);
  await checkThirdReviewPass(plan);
  const live = JSON.parse(await fs.readFile(f.config.curation));
  for (const [identity, entry] of Object.entries(f.curation.collections)) {
    if (identity === key(2)) assert.deepEqual(live.collections[identity], { name: entry.name, artist: entry.artist, group: "excluded", directory: entry.directory });
    else assert.deepEqual(live.collections[identity], entry);
  }
  const rejectedFolder = path.join("samples", f.curation.collections[key(2)].group, f.curation.collections[key(2)].directory);
  const after = await tree(f.directory);
  for (const [file, contents] of before) {
    if (file.startsWith(rejectedFolder + path.sep)) assert.equal(after.has(file), false);
    else if (!["curation.json", path.join("Development", "review-pass.json")].includes(file)) assert.deepEqual(after.get(file), contents, file);
  }
  assert.deepEqual(await fs.readFile(path.join(f.config.reviews, "pass-2.json")), f.secondSource);
  const repeated = await planThirdReviewPass({ ...f.config, source: undefined });
  assert.equal(await applyThirdReviewPass(repeated), 0);
  assert.deepEqual(await tree(f.directory), after);
  const logs = [];
  await main(["--check"], { ...f.config, source: undefined, log: value => logs.push(value) });
  assert.equal(JSON.parse(logs[0]).mode, "check");
});

test("foreign decisions, incomplete coverage, metadata mismatches, and source changes fail without mutation", async t => {
  const changes = [
    review => { review.passID = "pass-3"; },
    review => { review.version = 2; },
    review => { delete review.collections[key(4)]; },
    review => { review.collections[key(5)] = review.collections[key(4)]; },
    review => { review.collections[key(4)].decision = null; },
    review => { review.collections[key(4)].name = "Different"; },
    review => { delete review.collections[key(4)].note; },
  ];
  for (const change of changes) {
    const f = await fixture(t);
    change(f.second);
    const source = json(f.second);
    await fs.writeFile(f.config.source, source);
    const before = await tree(f.directory);
    await assert.rejects(planThirdReviewPass({ ...f.config, expectedSourceSHA256: digest(source) }));
    assert.deepEqual(await tree(f.directory), before);
  }
  const f = await fixture(t);
  await fs.appendFile(f.config.source, "\n");
  await assert.rejects(planThirdReviewPass(f.config), /SHA-256/u);
});

test("frozen archives and unrelated live curation changes cannot be overwritten", async t => {
  const f = await fixture(t);
  const archive = path.join(f.config.reviews, "pass-2.json");
  await fs.writeFile(archive, "immutable previous archive");
  await assert.rejects(planThirdReviewPass(f.config), /Refusing to replace frozen/u);
  await fs.rm(archive);
  await fs.writeFile(path.join(f.config.reviews, "pass-1-curation.json"), f.originalCuration);
  f.curation.collections[key(1)].name = "Unrelated update";
  await fs.writeFile(f.config.curation, json(f.curation));
  await assert.rejects(planThirdReviewPass(f.config), /unrelated changes/u);
});

test("unsafe or mismatched removal manifests fail before publication", async t => {
  const f = await fixture(t), entry = f.curation.collections[key(2)];
  const manifestPath = path.join(f.config.samplesRoot, entry.group, entry.directory, "manifest.json");
  const manifest = JSON.parse(await fs.readFile(manifestPath));
  manifest.identity = key(1);
  await fs.writeFile(manifestPath, json(manifest));
  const before = await tree(f.directory);
  await assert.rejects(planThirdReviewPass(f.config), /manifest differs/u);
  assert.deepEqual(await tree(f.directory), before);
});

test("inputs changing after validation prevent every publication and deletion", async t => {
  const f = await fixture(t), plan = await planThirdReviewPass(f.config);
  await fs.appendFile(f.config.source, "\n");
  const before = await tree(f.directory);
  await assert.rejects(applyThirdReviewPass(plan), /changed after validation/u);
  assert.deepEqual(await tree(f.directory), before);
});

test("publication failure restores live files and retains sample folders plus safe immutable archives", async t => {
  const f = await fixture(t), before = await tree(f.directory), plan = await planThirdReviewPass(f.config);
  const rename = fs.rename;
  let failed = false;
  fs.rename = async (source, destination) => {
    if (destination === f.config.runtimeOutput && !failed) { failed = true; throw new Error("Injected publication failure"); }
    return rename(source, destination);
  };
  try { await assert.rejects(applyThirdReviewPass(plan), /Injected publication failure/u); }
  finally { fs.rename = rename; }
  const after = await tree(f.directory);
  for (const [file, contents] of before) assert.deepEqual(after.get(file), contents, file);
  assert.deepEqual(await fs.readFile(path.join(f.config.reviews, "pass-1-curation.json")), f.originalCuration);
  assert.deepEqual(await fs.readFile(path.join(f.config.reviews, "pass-2.json")), f.secondSource);
  assert.equal([...after.keys()].some(file => file.endsWith(".tmp")), false);
  await applyThirdReviewPass(await planThirdReviewPass(f.config));
  await checkThirdReviewPass(await planThirdReviewPass(f.config));
});
